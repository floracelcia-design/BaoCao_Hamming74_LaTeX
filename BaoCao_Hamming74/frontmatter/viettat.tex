\chuongphu{DANH MỤC CÁC TỪ VIẾT TẮT}

\begin{center}
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{|p{2.8cm}|p{11.7cm}|}
\hline
\textbf{Từ viết tắt} & \textbf{Ý nghĩa}\\ \hline
ASCII   & American Standard Code for Information Interchange -- bảng mã chuẩn
          trao đổi thông tin của Hoa Kỳ\\ \hline
BER     & Bit Error Rate -- tỉ lệ lỗi bit\\ \hline
BSC     & Binary Symmetric Channel -- kênh đối xứng nhị phân\\ \hline
BTL     & Bài tập lớn\\ \hline
ĐHQG    & Đại học Quốc gia\\ \hline
ECC     & Error-Correcting Code -- mã sửa lỗi\\ \hline
GF(2)   & Galois Field of 2 elements -- trường hữu hạn hai phần tử $\{0, 1\}$\\ \hline
MC      & Monte Carlo -- phương pháp mô phỏng ngẫu nhiên\\ \hline
MSSV    & Mã số sinh viên\\ \hline
SEC-DED & Single Error Correction, Double Error Detection -- sửa $1$ lỗi, phát
          hiện $2$ lỗi\\ \hline
STT     & Số thứ tự\\ \hline
TP. HCM & Thành phố Hồ Chí Minh\\ \hline
UTF-8   & 8-bit Unicode Transformation Format -- chuẩn mã hóa ký tự thành
          byte\\ \hline
XOR     & Exclusive OR -- phép hoặc loại trừ (cộng modulo $2$)\\ \hline
$\oplus$ & Phép cộng modulo $2$ (XOR)\\ \hline
\end{tabular}
\end{center}
